predicate robotequipped {
    client - Agent
}

predicate activeTool {
    tool - Tool
}

    client - Agent
}

    agent - Agent,
    tool - Tool
}

    tool - Tool,
    loc - Location
}

    parameters {
        client: robot

        to: location

    precondition {
        PredicateInstance: atAgent(agent = client, location = from, isNegated = false)
    }    
    effect {
        PredicateInstance: atAgent(agent = client, location = from, isNegated = true)
        PredicateInstance: atAgent(agent = client, location = to, isNegated = false)
    }   
    function {
        {client, from, to},
        {Boolean}
    }
    implementation {
        travelImplementation
    }
}

// Robot stacks one object on top of another
Action stackML {
    parameters {
        obj1: element
        obj2: element
        client: robot
        vg: vacuumGripper
        pr: positionOnRail
    }
    precondition {
        PredicateInstance: vgempty(client = client, isNegated = true)
        PredicateInstance: holding(agent = client, myObject = obj1, isNegated = false)
        PredicateInstance: atAgent(agent = client, location = pr, isNegated = false)
        PredicateInstance: activeTool(tool = vg, isNegated = false)
        PredicateInstance: atplace(myObject = obj2, place = pr, isNegated = false)
        PredicateInstance: positionfree(pos = pr, isNegated = true)
    }    
    effect {
        PredicateInstance: ontop(myObject1 = obj1, myObject2 = obj2, isNegated = false)
        PredicateInstance: holding(agent = client, myObject = obj1, isNegated = true)
        PredicateInstance: atplace(myObject = obj1, place = pr, isNegated = false)
        PredicateInstance: vgempty(client = client, isNegated = false)
        PredicateInstance: clear(myObject = obj2, isNegated = true)
        PredicateInstance: clear(myObject = obj1, isNegated = false)
        PredicateInstance: stacked(myObject = obj1, isNegated = false)
    }   
    function {
        {obj1, obj2, client, vg, pr},
        {Boolean}
    }
    implementation {
        stackImplementation
    }
}
